    <script>
        // وظيفة التنقل بين الصفحات
        function showPage(pageId) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active-page'));
            document.getElementById(pageId).classList.add('active-page');
        }

        // إنشاء الـ 30 جزءاً
        const partsGrid = document.getElementById('partsGrid');
        for (let i = 1; i <= 30; i++) {
            let item = document.createElement('div');
            item.className = 'part-item';
            item.innerText = 'الجزء ' + i;
            item.onclick = () => openPartDetails(i);
            partsGrid.appendChild(item);
        }

        function openPartDetails(num) {
            document.getElementById('selectedPartTitle').innerText = 'الجزء ' + num;
            const modelsGrid = document.getElementById('modelsGrid');
            modelsGrid.innerHTML = '';
            
            // إنشاء 10 نماذج
            for (let j = 1; j <= 10; j++) {
                let mBox = document.createElement('div');
                mBox.className = 'box';
                mBox.innerText = j;
                mBox.onclick = () => showClips();
                modelsGrid.appendChild(mBox);
            }
            
            showPage('detailView');
            document.getElementById('clipsSection').style.display = 'none';
        }

        function showClips() {
            const clipsGrid = document.getElementById('clipsGrid');
            clipsGrid.innerHTML = '';
            document.getElementById('clipsSection').style.display = 'block';
            
            // إنشاء 6 مقاطع
            for (let k = 1; k <= 6; k++) {
                let cBox = document.createElement('div');
                cBox.className = 'box clip-box';
                cBox.innerText = 'مقطع ' + k;
                cBox.onclick = () => alert('هنا تظهر صفحة المصحف الملون للمقطع ' + k);
                clipsGrid.appendChild(cBox);
            }
        }
    </script>
